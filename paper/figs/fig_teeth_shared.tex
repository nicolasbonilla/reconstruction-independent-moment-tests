% fig_teeth_shared.tex -- native pgfplots. The SHARED-STATE teeth: both branches evaluated classically on the SAME
% subspace ground state |0_S> of H_S = Pi_S H Pi_S (Eq. app-leak, k=1), beside the idealized full-sector-GS curve.
% No shots and no device: the estimator needs only the sampled support and is classical post-processing; it is not a
% moment measured on a device. The support here is a deterministic truncation to the top-d configurations by |psi0|^2.
% L=12 doped Hubbard ring (nup=nd=4), U/t=8, sector Lanczos (run_teeth_shared.py -> teeth_shared.dat).
% y = relative discrepancy in %: |mhat_1 - mbar_1| / mhat_1 (columns rideal, rshared) and, since 2026-09-28,
% Delta_0 / mhat_0 (column leak).
% x = determinants kept d (log, REVERSED: full support left where curves ~0, truncation right where they saturate ~100%).
% The 5% line is an illustrative fixed threshold, not the deployed tau_1 rule.
% 2026-09-28 (review pass E: R1-01, V1, V3, V9): (i) added the zeroth-order shared-state residual
% Delta_0/mhat_0 = 1 - |Pi_S J|0_S>|^2 / |J|0_S>|^2 (column leak, run_teeth_shared.py l.104-107: the fraction of
% |J|0_S>|^2 outside S, i.e. the captured-weight deficit 1 - w), drawn in ink (not teal, which figstyle reserves for agreement), dash-dotted, with open triangle markers. In
% teeth_shared.dat it lies above the shared-state Delta_1 curve from d = 175758 down to d = 2200 (e.g. 4.02 vs 1.43 at
% d = 68744) and below it at full support and for d <= 1609, where both saturate (93.8 vs 99.4 at d = 337). No
% crossing marker is drawn for it: its 5% crossing is not in a committed file. (ii) The two 5% crossings (log-linear interpolation between computed d,
% run_teeth_shared.py crossing(); data/2026-08-26_teeth_shared_state.json first_crossing_5pct_*_d) are open crimson
% rings, not data-style dots. (iii) Markers drawn solid. (iv) Three-entry legend in the upper left, where every curve
% is below 30% for d > 2e4; it ends left of the steep rise (d < 8e3). Height lowered from 6.8 cm to 5.6 cm.
\begin{tikzpicture}
\begin{axis}[
  paperaxis, width=\columnwidth, height=5.6cm,
  xmode=log, x dir=reverse,
  xlabel={determinants kept $d$ \; (truncation $\rightarrow$)},
  ylabel={relative discrepancy (\%)},
  xmin=180, xmax=260000, ymin=-6, ymax=112,
  xtick={100000,10000,1000}, xticklabels={$10^5$,$10^4$,$10^3$},
  ytick={0,20,40,60,80,100},
  legend style={at={(0.03,0.98)}, anchor=north west, font=\scriptsize,
                draw=pMuted, fill=white, fill opacity=0.92, text opacity=1,
                row sep=0pt, inner sep=2pt},
  legend cell align=left, clip=false,
  axis on top=false,   % grid below data (on top, the light grid lines cut white breaks into the curves)
]
  % illustrative fixed threshold (not the deployed tau_1 rule), labelled directly
  \addplot[threshold, forget plot] coordinates {(260000,5) (180,5)};
  \node[pCrimson, font=\scriptsize, anchor=south east, inner sep=1.5pt] at (axis cs:200,5.8) {illustrative $5\%$ threshold};
  % idealized: independent estimator on the FULL-sector ground state
  \addplot[pIndigo, thick, densely dashed, mark=o, mark size=1.5pt,
           mark options={draw=pIndigo, line width=0.6pt, solid}]
      table[x=d,y=rideal]{figs/teeth_shared.dat};
  \addlegendentry{idealized $\Delta_1/\widehat m_1$}
  % shared state: both branches on the SAME subspace GS |0_S> (classical evaluation)
  \addplot[firesline] table[x=d,y=rshared]{figs/teeth_shared.dat};
  \addlegendentry{shared-state $\Delta_1/\widehat m_1$}
  % shared state, zeroth order: Delta_0/mhat_0 = captured-weight deficit (column leak)
  % ink, not teal: figstyle.tex reserves teal for agreement, and this curve is a residual
  \addplot[pInk, thick, dash dot, mark=triangle*, mark size=1.9pt,
           mark options={fill=white, draw=pInk, line width=0.6pt, solid}]
      table[x=d,y=leak]{figs/teeth_shared.dat};
  \addlegendentry{shared-state $\Delta_0/\widehat m_0$}

  % interpolated 5% crossings of the two Delta_1 curves: open crimson rings (values in the caption)
  \addplot[only marks, mark=o, mark size=3pt, pCrimson, line width=0.8pt, forget plot]
      coordinates {(43555,5) (15977,5)};
\end{axis}
\end{tikzpicture}
