% fig_hero.tex -- one subspace state, two operator evaluations (native TikZ, no PNG).
% 2026-10-02 (restructure): redrawn as the shared-state test the paper realizes. The sampled support S
% fixes the subspace ground state |0_S>; the reconstruction branch uses the operators restricted to S (A(omega) and
% mbar_k), the estimator branch the full sparse operators (mhat_k = <0_S|O_k|0_S>). Their difference is the leak of
% J|0_S> out of S (Proposition 1). Both branches are classical post-processing of the support; on hardware the device
% supplies S only. The decision is a leak score against a numerical tolerance (shared-state estimator) or tau_k
% (oracle calibration, Sec. 4); a pass is not a certificate.
\begin{tikzpicture}[font=\small]
  % --- nodes ---
  \node[pbox, fill=pGrid!40, minimum height=12mm] (S)
      at (0.9,2.2) {sampled support $\mathcal{S}$\\[1pt] subspace state $|0_\mathcal{S}\rangle$};
  \node[pbox, draw=pInk, fill=white, minimum height=10mm] (REC)
      at (5.6,3.5) {operators restricted to $\mathcal S$\\ $A(\omega)$, \ $\bar m_k=\int\omega^kA\,d\omega$};
  \node[pbox, draw=pIndigo, fill=pIndigo!8, minimum height=10mm] (MOM)
      at (5.6,0.9) {full operators\\ $\widehat m_k=\langle 0_\mathcal{S}|O_k|0_\mathcal{S}\rangle$};
  \node[pbox, draw=pInk, fill=white, minimum height=10mm] (CMP)
      at (10.2,2.2) {$\Delta_k=\lvert\widehat m_k-\bar m_k\rvert$\\ $=$ leak out of $\mathcal S$};
  \node[pbox, draw=pTeal, fill=pTeal!10, minimum height=9mm] (CORR)
      at (14.4,3.4) {\textbf{pass}\\ below threshold\\[-1pt] {\scriptsize\itshape (not a certificate)}};
  \node[pbox, draw=pOx, fill=pOx!8, minimum height=9mm] (FALS)
      at (14.4,1.0) {\textbf{reject}\\ above threshold};

  % --- flow arrows ---
  \draw[pflow] (S) to[out=60,in=180]  (REC.west);
  \draw[pflow] (S) to[out=-60,in=180] (MOM.west);
  \draw[pflow] (REC.east) to[out=0,in=120]  (CMP.north west);
  \draw[pflow] (MOM.east) to[out=0,in=-120] (CMP.south west);
  \draw[pflow] (CMP.east) to[out=60,in=180]  (CORR.west);
  \draw[pflow] (CMP.east) to[out=-60,in=180] (FALS.west);

  % --- forbidden path: A(w) is NEVER read back into the moment branch ---
  \draw[pforbid] (REC.south) -- (MOM.north)
      node[midway] (X) {};
  \node[circle, draw=pOx, line width=0.9pt, minimum size=4.2mm, inner sep=0pt] at (X) {};
  \draw[pOx, line width=0.9pt] ($(X)+(-0.15,-0.15)$) -- ($(X)+(0.15,0.15)$);
  \node[font=\scriptsize\itshape, pMuted, align=center, anchor=east] at ($(X)+(-0.45,0)$)
      {never read\\ back off $A(\omega)$};

  % 2026-10-02: the header (classical evaluation) and footer (O_k, e0, threshold) lines, set at
  % about 5 pt after scaling, are removed; the caption states all three.
\end{tikzpicture}
